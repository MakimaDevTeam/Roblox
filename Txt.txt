local function canUseFileSystem()
    return type(isfile) == "function"
        and type(readfile) == "function"
        and type(writefile) == "function"
end

local function ensureFolder()
    if type(makefolder) ~= "function" then
        return
    end

    if type(isfolder) == "function" then
        if not isfolder(KEY_FOLDER) then
            pcall(function()
                makefolder(KEY_FOLDER)
            end)
        end
    else
        pcall(function()
            makefolder(KEY_FOLDER)
        end)
    end
end

local function saveKey(key)
    if not canUseFileSystem() then
        return false
    end

    ensureFolder()

    local ok = pcall(function()
        writefile(KEY_FILE, key)
    end)

    return ok
end

local function getSavedKey()
    if not canUseFileSystem() then
        return nil
    end

    if type(isfile) ~= "function" then
        return nil
    end

    local exists = false

    pcall(function()
        exists = isfile(KEY_FILE)
    end)

    if not exists then
        return nil
    end

    local ok, content = pcall(function()
        return readfile(KEY_FILE)
    end)

    if ok and content then
        content = tostring(content):gsub("^%s+", ""):gsub("%s+$", "")

        if content ~= "" then
            return content
        end
    end

    return nil
end

local function deleteSavedKey()
    if type(delfile) ~= "function" then
        return
    end

    pcall(function()
        if isfile(KEY_FILE) then
            delfile(KEY_FILE)
        end
    end)
end

local Junkie

do
    local ok, result = pcall(function()
        return loadstring(
            game:HttpGet("https://jnkie.com/sdk/library.lua")
        )()
    end)

    if not ok or not result then
        warn("[Makima Scripts] Não foi possível carregar o JNKIE.")
        return
    end

    Junkie = result
end

Junkie.service = JNKIE_SERVICE
Junkie.identifier = JNKIE_IDENTIFIER
Junkie.provider = JNKIE_PROVIDER

local function validateKey(key)
    if not key or key == "" then
        return false
    end

    local ok, result = pcall(function()
        return Junkie.check_key(key)
    end)

    if not ok or not result then
        return false
    end

    return result.valid == true
end

        local ok, source = pcall(function()
            return game:HttpGet(ORIGINAL_SCRIPT)
        end)

        if not ok or not source then
            warn("[Makima Scripts] Não foi possível buscar o Parkour original.")
            return nil
        end

        local tableText = source:match(
            "local hitboxData%s*=%s*(%b{})"
        )

        if not tableText then
            warn("[Makima Scripts] hitboxData não encontrada.")
            return nil
        end

        local loadOk, data = pcall(function()
            return loadstring("return " .. tableText)()
        end)

        if loadOk and type(data) == "table" then
            return data
        end

        return nil
    end

    local function CreateHitboxes()

        if #Hitboxes > 0 then
            return
        end

        local hitboxData = getOriginalHitboxData()

        if not hitboxData then
            return
        end

        for _, v in ipairs(hitboxData) do

            local p = Instance.new("Part")

            p.Name = "MakimaScripts_Hitbox"
            p.Size = v.size
            p.Position = v.pos
            p.Anchored = true
            p.Transparency = 1
            p.CanCollide = false
            p.Parent = workspace

            local box = Instance.new("SelectionBox")

            box.Adornee = p
            box.Color3 = Color3.fromRGB(0, 100, 255)
            box.SurfaceColor3 = Color3.fromRGB(0, 150, 255)
            box.SurfaceTransparency = 0.5
            box.LineThickness = 0.08
            box.Parent = p

            table.insert(Hitboxes, p)
        end
    end

    local function ClearHitboxes()

        for _, v in pairs(Hitboxes) do
            pcall(function()
                v:Destroy()
            end)
